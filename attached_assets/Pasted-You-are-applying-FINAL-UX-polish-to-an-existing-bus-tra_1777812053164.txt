const crowdOrder = {
  "Low": 1,

const sortedBuses = useMemo(() => {
  return [...buses].sort((a, b) => {
    const getEtaValue = (bus) => {
      if (bus.distanceToNextStop < 30 || bus.eta < 2) return 0;
      return bus.eta ?? 999;
    };

    const etaDiff = getEtaValue(a) - getEtaValue(b);
    if (etaDiff !== 0) return etaDiff;

    const aCrowd = crowdOrder[a.crowdLevel] ?? 2;
    const bCrowd = crowdOrder[b.crowdLevel] ?? 2;
    return aCrowd - bCrowd;
  });
}, [buses]);
