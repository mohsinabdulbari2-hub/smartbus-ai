2️⃣ FREQUENCY ENGINE (ALL 4 COMBINED)

Goal

Compute realistic bus frequency using:

- backend baseline
- live ETA gaps

---

Step 1 — Time slot

function getTimeSlot(hour) {
  if (hour >= 6 && hour < 10) return "6-10";
  if (hour >= 11 && hour < 15) return "11-15";
  if (hour >= 16 && hour < 20) return "16-20";
  return "21+";
}

---

Step 2 — Filter buses for stop

function getBusesForStop(buses, stopIndex) {
  return buses.filter(
    b =>
      b.nextStopIndex != null &&
      b.nextStopIndex >= stopIndex &&
      b.nextStopIndex <= stopIndex + 2
  );
}

---

Step 3 — Live frequency

function getLiveFrequency(etas) {
  if (!etas || etas.length < 2) return null;

  const sorted = [...etas].sort((a, b) => a - b);
  const gaps = [];

  for (let i = 1; i < sorted.length; i++) {
    const gap = sorted[i] - sorted[i - 1];
    if (gap > 0) gaps.push(gap);
  }

  if (!gaps.length) return null;

  const avg = gaps.reduce((a, b) => a + b, 0) / gaps.length;
  return Math.round(3600 / avg);
}

---

Step 4 — Final fusion

function getFinalFrequency({
  routeId,
  directionId,
  stopIndex,
  buses,
  baseline
}) {
  const hour = new Date().getHours();
  const slot = getTimeSlot(hour);

  const key = `${routeId}_${directionId}`;
  const base = baseline[key]?.[slot] ?? 6;

  const relevant = getBusesForStop(
    buses.filter(b => b.directionId === directionId),
    stopIndex
  );

  const etas = relevant.map(b => b.eta).filter(Boolean);
  const live = getLiveFrequency(etas);

  if (!live) return base;

  const weight =
    etas.length >= 4 ? 0.5 :
    etas.length >= 2 ? 0.35 :
    0.2;

  return Math.round(base * (1 - weight) + live * weight);
}
