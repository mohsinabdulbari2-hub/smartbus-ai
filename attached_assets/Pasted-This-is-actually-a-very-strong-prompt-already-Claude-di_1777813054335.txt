useEffect(() => {
  if (!trackedBusId || !mapRef.current) return;

  const trackedBus = buses.find(b => b.id === trackedBusId);
  if (!trackedBus) return;

  const newPos = {
    latitude: trackedBus.lat,
    longitude: trackedBus.lng,
  };

  const prevPos = lastCameraPosition.current;

  // 🚫 Skip if movement is too small (prevents jitter)
  if (prevPos) {
    const dx = Math.abs(prevPos.latitude - newPos.latitude);
    const dy = Math.abs(prevPos.longitude - newPos.longitude);

    if (dx < 0.0001 && dy < 0.0001) {
      return; // ignore tiny movement
    }
  }

  lastCameraPosition.current = newPos;

  if (!hasInitialZoomed) {
    mapRef.current.animateCamera({
      center: newPos,
      zoom: 15,
    }, { duration: 800 });

    setHasInitialZoomed(true);
  } else {
    mapRef.current.animateCamera({
      center: newPos,
    }, { duration: 600 });
  }

}, [buses, trackedBusId]);
